check_end_aura_win_condition_mod = {
	export_trigger_value_to_variable = {
		trigger = count_systems_with_aura
		variable = nb_conquered_systems
	}
	if = {
		limit = {
			count_end_cycle_systems <= nb_conquered_systems
			NOT = {
				any_country = {
					is_fallen_empire = yes
				}
			}
		}
		if = {
			limit = { has_origin = origin_endbringers }
			country_event = { id = shroud.8600 }
		}
		else = {
			country_event = { id = shroud.4970 }
		}
	}
}

		name = "NAME_Shroud_science"
		settings = {
			can_upgrade = no
			can_change_composition = no
			can_change_leader = no
			can_disband = no
			spawn_debris = no
		}
		effect = {
			set_owner = root.owner
			create_ship_design = {
				design = NAME_Shroud_science
			}
			create_ship = {
				design = last_created_design
				name = "NAME_Shroud_science"
				prefix = no
				upgradable = no
				effect = {
					set_ship_flag = is_shroud_science_ship
					set_ship_flag = special_science_ship
				}
			}
			set_fleet_stance = evasive
			set_location = root.capital_scope
			assign_leader = last_created_leader
		}
	}
}
