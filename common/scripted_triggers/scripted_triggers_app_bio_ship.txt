
is_ship_app_bio_mod = {
    OR = {
        is_ship_size = cosmo_crisis_mauler
        is_ship_size = cosmo_crisis_weaver
        is_ship_size = cosmo_crisis_harbinger
        is_ship_size = cosmo_crisis_stinger
        is_ship_size = crisis_mauler_mod
        is_ship_size = crisis_harbinger_mod
        is_ship_size = crisis_stinger_mod
		is_ship_size = behemoth_mod
    }
}


country_uses_bio_ships = {
    OR = {
        AND = {
            exists = this
	        NOT = { uses_ship_category = bio_ship }
            exists = prev
            prev = {
                OR = {
                    is_scope_type = ship_growth_stage
				    is_scope_type = ship
				    #is_scope_type = fleet
                }
                is_ship_app_bio_mod = yes  
            }
        }
        AND = {
            exists = this
	        uses_ship_category = bio_ship
        }
    }
}
