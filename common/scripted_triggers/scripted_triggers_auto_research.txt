is_shroud_terraformable_pc = {
	OR = {
		is_planet_class = pc_barren
		is_planet_class = pc_barren_cold
		is_planet_class = pc_shrouded
	}
	planet_size >= 10
	NOT = { has_planet_flag = terraform_nucleus_transfixed }
	NOT = { has_planet_flag = terraform_shroud_nucleus_transfixed }
}

has_shroud_science_in_orbit = {
	any_fleet_in_orbit = {
		any_owned_ship = {
			is_ship_size = shroud_science
		}
	}
}
