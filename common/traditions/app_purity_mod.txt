tr_purity_adopt = {
	possible = {
		custom_tooltip = {
			fail_text = REQUIRES_FINISHED_EVOLUTION
			has_country_flag = purity_tradition_unlocked
		}
	}
	unlocks_agenda = agenda_purity_perfection
	modifier = {
		leader_lifespan_mult = 0.2
	}
	on_enabled = {
		hidden_effect = {
			set_country_flag = purity_unlock_level_1
			set_country_flag = purity_unlock_level_2
			set_country_flag = purity_unlock_level_3
			set_country_flag = genetic_ascension_stage_1_purity
			set_country_flag = genetic_ascension_stage_2_purity
			set_country_flag = genetic_ascension_stage_3_purity
			remove_modifier = purity_unlock_level_1
			add_modifier = {
				modifier = purity_unlock_level_2
			}
		}
	}
}

tr_purity_finish = {
	custom_tooltip_with_modifiers = tr_purity_finish_tt
	modifier = {
		ascension_perks_add = 1
	}

	on_enabled = {
		hidden_effect = {
			country_event = { id = timeline.80 }
			if = {
				limit = {
					is_hive_empire = no
					aap_is_advanced_government_mod = no
					NOT = { has_origin = origin_evolutionary_predators }
					NOT = { has_origin = origin_clone_army }
				}
				country_event = {
					id = bio.170
				}
			}
			else_if = {
				limit = {
					is_hive_empire = yes
				}
				country_event = {
					id = bio.185
				}
			}
			luminary_speech_effect = yes
		}
	}

	tradition_swap = {
		# the wilderness recives no alternate authority
		trigger = {
			is_wilderness_empire = yes
		}
		custom_tooltip_with_modifiers = tr_purity_finish_wilderness_tt
		modifier = {
			ascension_perks_add = 1
		}
	}
}
