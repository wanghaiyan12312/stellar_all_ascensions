planet_event = {
	id = leadershroud.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_paragon_dlc = yes
		has_shroud_science_in_orbit = yes
		NOT = { has_planet_flag = terraform_nucleus_transfixed }
		NOT = { has_planet_flag = terraform_nucleus_transfix_started }
		NOT = { has_planet_flag = terraform_shroud_nucleus_transfixed }
	}

	immediate = {
		planet_event = { id = leadershroud.3 days = 3 }

		from.owner = {
			set_timed_country_flag = {
				flag = terraform_nucleus_purchase
				days = 3600
			}
			set_saved_date = {
				key = leadershroud_terraform_nucleus_timer
				days_from_present = 3600
				expires = 3600
			}
			# Notification - Terraforming Started
			create_message = {
				type = MESSAGE_SHROUD_TERRAFORM_STARTED
				localization = MESSAGE_SHROUD_TERRAFORM_STARTED_DESC
				days = 10
				target = prev
				variable = {
					type = name
					localization = PLANET
					scope = this
				}
			}
		}

		remove_planet_flag = terraform_nucleus_transfix_canceled
		set_planet_flag = terraform_nucleus_transfix_started

		# Set vars to adapt effect to planet size
		export_trigger_value_to_variable = {
			trigger = planet_size
			variable = effect_size_var
		}
		multiply_variable = {
			which = effect_size_var
			value = 0.13
		}
		create_ambient_object = {
			type = terraforming_planet_effect
			location = this
			use_3d_location = yes
			entity_scale_to_size = yes
			scripted_scale = effect_size_var
		}
		last_created_ambient_object = {
			set_ambient_object_flag = shroud_science_terraforming_effect
		}
		# Set vars to adapt beam to planet size
		export_trigger_value_to_variable = {
			trigger = planet_size
			variable = beam_size_var
		}
		switch = {
			trigger = planet_size
			30 >= { multiply_variable = { which = beam_size_var value = 0.26 } }
			28 >= { multiply_variable = { which = beam_size_var value = 0.27 } }
			26 >= { multiply_variable = { which = beam_size_var value = 0.28 } }
			24 >= { multiply_variable = { which = beam_size_var value = 0.30 } }
			22 >= { multiply_variable = { which = beam_size_var value = 0.31 } }
			20 >= { multiply_variable = { which = beam_size_var value = 0.34 } }
			19 >= { multiply_variable = { which = beam_size_var value = 0.36 } }
			18 >= { multiply_variable = { which = beam_size_var value = 0.37 } }
			16 >= { multiply_variable = { which = beam_size_var value = 0.42 } }
			14 >= { multiply_variable = { which = beam_size_var value = 0.47 } }
			12 >= { multiply_variable = { which = beam_size_var value = 0.58 } }
			10 >= { multiply_variable = { which = beam_size_var value = 0.65 } }
		}
		create_ambient_object = {
			type = terraforming_ship_beam
			location = From
			entity_face_object = this
			use_3d_location = yes
			entity_scale_to_size = yes
			scripted_scale = beam_size_var
			play_animation_once = yes
		}
		last_created_ambient_object = {
			set_ambient_object_flag = shroud_science_terraforming_beam
		}
	}
}

planet_event = {
	id = leadershroud.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_paragon_dlc = yes
		has_shroud_science_in_orbit = yes
		NOT = { has_planet_flag = terraform_nucleus_transfix_canceled }

		NOT = { has_planet_flag = terraform_shroud_nucleus_transfixed }
	}

	immediate = {
		# Destroy beam effect
		solar_system = {
			every_system_ambient_object = {
				limit = {
					OR = {
						has_ambient_object_flag = shroud_science_terraforming_beam
					}
				}
				destroy_ambient_object = this
			}
		}

		set_planet_flag = terraform_nucleus_transfixed
		set_planet_flag = terraform_shroud_nucleus_transfixed
		remove_planet_flag = terraform_nucleus_transfix_started
		start_terraform_process = pc_gaia
	}

	after = {
	}
}

# Terraformation completed

# This = Planet
# From = Terraforming country
planet_event = {
	id = leadershroud.4
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_paragon_dlc = yes
		has_planet_flag = terraform_shroud_nucleus_transfixed
	}

	immediate = {
		# Add shroud-specific modifier to planet and remove original
		remove_modifier = terraformation_nucleus_side_effect
		add_modifier = { modifier = shroud_terraforming_effect }

		# Destroy planet effect
		solar_system = {
			every_system_ambient_object = {
				limit = {
					OR = {
						has_ambient_object_flag = shroud_science_terraforming_effect
					}
				}
				destroy_ambient_object = this
			}
		}

		remove_planet_flag = terraform_nucleus_transfixed
		remove_planet_flag = terraform_shroud_nucleus_transfixed
	}
}

planet_event = {
	id = leadershroud.5
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_paragon_dlc = yes
	}

	immediate = {
		set_planet_flag = terraform_nucleus_transfix_canceled
		remove_planet_flag = terraform_nucleus_transfix_started

		# Destroy beam and planet effects
		solar_system = {
			every_system_ambient_object = {
				limit = {
					OR = {
						has_ambient_object_flag = shroud_science_terraforming_beam
						has_ambient_object_flag = shroud_science_terraforming_effect
					}
				}
				destroy_ambient_object = this
			}
		}

		solar_system.owner = {
			# Notification - Terraforming Canceled
			create_message = {
				type = MESSAGE_SHROUD_TERRAFORM_CANCELED
				localization = MESSAGE_SHROUD_TERRAFORM_CANCELED_DESC
				days = 10
				target = this
				variable = {
					type = name
					localization = PLANET
					scope = this
				}
			}
		}

		
	}
}

country_event = {
	id = leadershroud.6
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		is_country_type = default
		from = { 
			has_leader_flag = leader_shroud_legendary_mod
		}
	}

	immediate = {
		if = {
			limit = {
				from = {
					has_leader_flag = legendary_govern_scientist_mod
				} 
			}
			clone_leader = {
				target = from
				effect = {
					set_leader_flag = immune_to_negative_traits
					set_leader_flag = leader_shroud_legendary_mod
					set_leader_flag = legendary_govern_scientist_mod
					remove_all_negative_traits = yes
					update_legendary_govern_scientist = yes
				}			
	    	}
			add_shroud_ship = yes
		}else = {
			clone_leader = {
				target = from
				effect = {
					set_leader_flag = immune_to_negative_traits
					remove_all_negative_traits = yes
				}			
	    	}
		}
		from = {
			kill_leader = { show_notification = no }
		}
	}

	after = {
		country_event = {
			id = leadershroud.8
			days = 10
		}
	}
}
